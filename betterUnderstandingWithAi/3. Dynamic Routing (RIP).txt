3. Dynamic Routing (RIP)

router rip
# Enable RIP Routing

version 2
# Use RIP Version 2

no auto-summary
# Disable automatic summarization

network <CONNECTED_NETWORK_1>
# Advertise Connected Network 1

network <CONNECTED_NETWORK_2>
# Advertise Connected Network 2

...
# Add one 'network' command for every directly connected network.


// practice 

//router 1______________
en
conf t 

int gig0/0
ip add 192.168.30.1 255.255.255.0
no shut 

interface gig0/1
ip address 192.168.50.1 255.255.255.0
no shutdown

ip dhcp pool net1
network 192.168.30.0 255.255.255.0
default-router 192.168.30.1
dns-server 8.8.8.8

router rip
version 2
no auto summary 
network 192.168.30.0
network 192.168.50.0
